{{- /* Render a single frontmatter value, linking if possible. Receives dict: val, key, titleIndex, noLink, taxonomyKeys */ -}} {{- $val := .val -}} {{- $key := .key -}} {{- $titleIndex := .titleIndex -}} {{- $noLink := .noLink -}} {{- $taxonomyKeys := .taxonomyKeys -}} {{- $valType := printf "%T" $val -}} {{- if eq $valType "bool" -}} {{ $val }} {{- else if strings.HasSuffix $valType "Time" -}} {{ $val | time.Format "2006-01-02" }} {{- else -}} {{- $str := printf "%v" $val -}} {{- $isDateStr := findRE `^\d{4}-\d{2}-\d{2}` $str -}} {{- if and (len $isDateStr) (strings.HasPrefix $key "date") -}} {{- $dateStr := substr $str 0 10 -}} {{ $dateStr }} {{- else -}} {{- $taxBase := index $taxonomyKeys $key -}} {{- if $taxBase -}} {{ $str }} {{- else if $noLink -}} {{ $str | humanize }} {{- else -}} {{- $url := index $titleIndex (lower $str) -}} {{- if $url -}} {{ $str | humanize }} {{- else -}} {{ $str | humanize }} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
